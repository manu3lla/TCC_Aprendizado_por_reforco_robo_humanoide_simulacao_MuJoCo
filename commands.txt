-----------------------------------
FACULDADE - Computador 1 - treinar v3 frente certa com passo largo seguro:

python reinforcement_learning/otimizar_hpps_optuna.py \
  --somente-treinar-melhor \
  --device cuda \
  --modo-gait-andando \
  --continuar-de reinforcement_learning/melhores_modelos/treino_gait_andando_v3_frente_certa_PROTEGIDO_NAO_APAGAR/best_model.zip \
  --final-timesteps 2000000 \
  --eval-freq 50000 \
  --n-eval-episodes 5 \
  --seed 42 \
  --final-run-name treino_v3_passo_largo_seguro_v1
